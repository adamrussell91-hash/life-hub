export function getLowpolyTreeArchetype(id) {
  return LOWPOLY_TREE_ARCHETYPES[id] || null
}

export function createLowpolyTreeArchetypeParams(id, overrides = {}) {
  const entry = getLowpolyTreeArchetype(id)
  if (!entry) throw new RangeError(`Unknown low-poly tree archetype: ${id}`)
  return { ...entry.params, ...overrides }
}
