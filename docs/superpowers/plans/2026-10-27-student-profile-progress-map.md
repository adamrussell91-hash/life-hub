# Student Profile Progress Map Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A student opens `/s/classes/<class>/profile`, sees the unit map greyed out, and it fills in colour as they finish lessons and show consistent top-end mastery, with no login and no new student data.

**Architecture:** Identity reuses the existing device claim (`class-claims.mjs`). A new public endpoint `POST /api/published/profile/:classId` resolves the claim, reads create-only "done" records under `profile/{class}/{student}/done/{lesson}`, runs a pure mastery function over the student's markbook/quiz evidence, and returns only the map and four bar fills (no scores, bands or counts). A new **Done** block writes the done records. Spec: `docs/superpowers/specs/2026-10-27-student-profile-progress-map-design.md`. Read it first.

**Tech Stack:** Netlify functions (`.mjs`, Blobs store), TypeScript + Vite app in `apps/teaching`, zod, vitest (`apps/teaching/tests/unit`), `node --test` (`tests/unit/*.test.js`), Playwright (`tests/browser`), esbuild bundles generated by `scripts/generate-alchemy-schema.mjs`.

**Mandatory gate:** `npm run pre-pr-check` must exit 0 before opening or updating the PR (`CLAUDE.md`). Run `npm ci` first in a fresh container, or every test fails with `ERR_MODULE_NOT_FOUND`. Open the PR ready for review, not draft, and enable auto-merge only after Adam has approved the work.

```ts
import { rungForMark } from '../skills/model';
import { BANDS, MASTERY } from './constants';
import type { Evidence } from './types';

export type MasteryGroup = 'understanding' | 'responding' | 'concepts' | 'skills';
export type MasteryItem = { group: MasteryGroup; tag: string };
export type MasteryLevel = 'none' | 'building' | 'mastered';
/** Fractions of a group's items, 0–1. `building` excludes mastered. Never counts or scores. */
export type MasteryFill = { mastered: number; building: number };

export const MASTERY_GROUPS: MasteryGroup[] = ['understanding', 'responding', 'concepts', 'skills'];

/**
 * One student's level on one item. A checkpoint is one source (a quiz or an assessed piece),
 * so two good questions in the same quiz are one checkpoint.
 * - Syllabus points and concepts: a checkpoint is top-end when it has at least BANDS.minQuestions
 *   tagged questions averaging BANDS.secure or more.
 * - Skills: a checkpoint is top-end at Accomplished or Excellent; any rung from Capable up is building.
 */
export function levelFor(group: MasteryGroup, tag: string, evidence: readonly Evidence[]): MasteryLevel {
  const bySource = new Map<string, number[]>();
  for (const e of evidence) {
    if (!e.tags.includes(tag)) continue;
    bySource.set(e.source, [...(bySource.get(e.source) ?? []), e.share]);
  }
  let top = 0;
  let reached = 0;
  for (const shares of bySource.values()) {
    const mean = shares.reduce((sum, x) => sum + x, 0) / shares.length;
    if (group === 'skills') {
      const rung = rungForMark(mean, 1);
      if (rung === 'accomplished' || rung === 'excellent') top += 1;
      if (rung !== 'below') reached += 1;
    } else if (shares.length >= BANDS.minQuestions && mean >= BANDS.secure) {
      top += 1;
      reached += 1;
    }
  }
  if (top >= MASTERY.checkpoints) return 'mastered';
  return reached >= 1 ? 'building' : 'none';
}

const round2 = (n: number): number => Math.round(n * 100) / 100;

/** The four bars for one student. `evidence` must already be this student's only. */
export function fillsFor(items: readonly MasteryItem[], evidence: readonly Evidence[]): Record<MasteryGroup, MasteryFill> {
  const out = {} as Record<MasteryGroup, MasteryFill>;
  for (const group of MASTERY_GROUPS) {
    const mine = items.filter((i) => i.group === group);
    if (!mine.length) {
      out[group] = { mastered: 0, building: 0 };
      continue;
    }
    const levels = mine.map((i) => levelFor(group, i.tag, evidence));
    out[group] = {
      mastered: round2(levels.filter((l) => l === 'mastered').length / mine.length),
      building: round2(levels.filter((l) => l === 'building').length / mine.length)
    };
  }
  return out;
}
```

```js
/**
 * Student profile: done records and the map payload for one claimed student.
 *
 *   profile/{class}/{student}/done/{lesson}   { at, via }   create-only
 *
 * The student is whoever this device has claimed (class-claims.mjs). Nothing here stores a
 * name, and the payload returns only the student's own map and four bar fills.
 */
import { isSafeId } from './class-examples.mjs';
import { claimFor, loadRoster } from './class-claims.mjs';
import { listBlobKeys } from './blobs-list.mjs';
import { read, reject } from './question-drop.mjs';
import { deleteKey, draftLessonKey, unitKey } from './teaching-blobs.mjs';
import { markbookAction } from './markbook.mjs';
import { fillsFor, markbookEvidence } from './profile-evidence.mjs';
import { profileItems } from './profile-items.mjs';

const doneKey = (classId, studentId, lessonId) => `profile/${classId}/${studentId}/done/${lessonId}`;
const donePrefix = (classId, studentId) => `profile/${classId}/${studentId}/done/`;
const ROLES = new Set(['core', 'support', 'extension', 'optional']);

export async function recordDone(store, classId, studentId, lessonId, via) {
  if (!isSafeId(lessonId)) reject(400, 'bad_request', 'Unknown lesson.');
  await store.setJSON(doneKey(classId, studentId, lessonId), { at: new Date().toISOString(), via }, { onlyIfNew: true });
}

export async function listDone(store, classId, studentId) {
  const prefix = donePrefix(classId, studentId);
  const out = {};
  await Promise.all((await listBlobKeys(store, prefix)).map(async (key) => {
    const held = await read(store, key);
    if (held) out[key.slice(prefix.length)] = held;
  }));
  return out;
}

export async function removeDone(store, classId, studentId, lessonId) {
  if (!isSafeId(lessonId)) reject(400, 'bad_request', 'Unknown lesson.');
  await deleteKey(store, doneKey(classId, studentId, lessonId));
}

/** The class's current unit, lessons in unit order with their route role. */
async function unitLessons(store, cls) {
  const unit = cls.current_unit_id ? await read(store, unitKey(cls.current_unit_id)) : null;
  const lessons = [];
  for (const id of unit?.lesson_ids ?? []) {
    const lesson = await read(store, draftLessonKey(id));
    if (lesson && lesson.title && !lesson.deleted_at) lessons.push(lesson);
  }
  return { unit, lessons };
}

export async function profileAction(store, classId, body) {
  if (!body || typeof body !== 'object' || Array.isArray(body)) reject(400, 'bad_request', 'Invalid request.');
  const action = body.action ?? 'status';
  if (!['status', 'done'].includes(action)) reject(400, 'bad_action', 'Unknown action.');
  const roster = await loadRoster(store, classId);
  if (!roster) reject(404, 'not_found', 'Class not found.');
  const student = await claimFor(store, roster, body.class_token);
  if (!student) {
    if (action === 'done') reject(403, 'need_initials', 'Type your initials first.');
    return { need: 'initials' };
  }
  if (action === 'done') {
    await recordDone(store, classId, student.id, body.lesson_id, 'tick');
    return profileFor(store, roster, student);
  }
  return profileFor(store, roster, student);
}

async function profileFor(store, roster, student) {
  const classId = roster.cls.id;
  const { lessons } = await unitLessons(store, roster.cls);
  const view = await markbookAction(store, classId, { action: 'load' });
  const done = await listDone(store, classId, student.id);
  // A submitted quiz counts as done without a tick.
  for (const quiz of view.quizzes) {
    if (quiz.scores?.[student.id]?.submitted_at && !done[quiz.lesson_id]) done[quiz.lesson_id] = { via: 'quiz' };
  }
  const took = new Set(view.markbook.took?.[student.id] ?? []);
  const rows = lessons.map((l) => {
    const role = ROLES.has(l.route_role) ? l.route_role : 'core';
    return {
      id: l.id,
      title: l.title,
      role,
      kind: view.quizzes.some((q) => q.lesson_id === l.id) ? 'quiz' : 'lesson',
      done: Boolean(done[l.id]),
      assigned: role === 'core' || took.has(l.id)
    };
  });
  const next = rows.find((r) => r.role === 'core' && !r.done)?.id ?? null;
  const evidence = markbookEvidence(view, []).evidence.filter((e) => e.student === student.id);
  const items = await profileItems(store, roster, lessons, view);
  return { need: null, initials: student.initials, map: { lessons: rows, next }, fills: fillsFor(items, evidence) };
}
```

```js
import { createPublicStudentHandler } from './_shared/public-student-gate.mjs';
import { profileAction } from './_shared/profile.mjs';
import { QuizError } from './_shared/question-drop.mjs';
import { errorResponse, guardRequestOrigin, okResponse, withCors, methodNotAllowed } from './_shared/http.mjs';
export const config = { path: '/api/published/profile/:id', rateLimit: { action: 'rate_limit', aggregateBy: ['ip', 'domain'], windowLimit: 600, windowSize: 60 } };
const MAX_BODY = 4000;
export function createPublishedProfileHandler(deps = {}) {
  return createPublicStudentHandler(async (request, context) => {
    const { env, store } = context;
    if (request.method !== 'POST') return withCors(methodNotAllowed('POST, OPTIONS'), request, env);
    const originError = guardRequestOrigin(request, env);
    if (originError) return withCors(originError, request, env);
    if (Number(request.headers.get('content-length')) > MAX_BODY) return withCors(errorResponse(413, 'too_large', 'That request is too large.', false), request, env);
    const match = new URL(request.url).pathname.match(/\/published\/profile\/([^/]+)$/);
    try {
      const result = await profileAction(store, match?.[1], await request.json());
      return withCors(okResponse(200, result, { 'cache-control': 'no-store' }), request, env);
    } catch (e) {
      if (e instanceof QuizError || e instanceof SyntaxError) return withCors(errorResponse(e.status ?? 400, e.code ?? 'bad_request', e instanceof SyntaxError ? 'Invalid request.' : e.message, false), request, env);
      throw e;
    }
  }, deps);
}
export default createPublishedProfileHandler();
```

```js
export async function teacherProfileAction(store, classId, body) {
  const roster = await loadRoster(store, classId);
  if (!roster) reject(404, 'not_found', 'Class not found.');
  const action = body?.action ?? 'list';
  if (!['list', 'remove'].includes(action)) reject(400, 'bad_action', 'Unknown action.');
  if (action === 'remove') {
    if (typeof body.student_id !== 'string' || !STUDENT_ID_PATTERN.test(body.student_id)) reject(400, 'bad_request', 'Unknown student.');
    await removeDone(store, classId, body.student_id, body.lesson_id);
  }
  const done = {};
  for (const s of roster.students) done[s.id] = await listDone(store, classId, s.id);
  return { done };
}
```
