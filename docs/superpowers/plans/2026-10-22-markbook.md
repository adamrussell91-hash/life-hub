# Class Markbook Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Each Teaching class gets a teacher-only markbook. Quiz scores fill in by themselves. The teacher adds assessment, hand-in and check columns. Assessments give a weighted overall mark. An outcomes view shows each student's evidence per syllabus outcome, and quiz insight flags reteaching needs and students to watch.

**Architecture:** Teacher columns and marks are stored in one Blobs record per class, `markbook/{classId}`. Quiz columns are never stored: on every load, the server finds the quiz lessons in the class's active units and scores each student's first handed-in attempt with the quiz module's own `scoreAttempt`. Validation lives in one TypeScript schema module, which is generated to `.mjs` for the server. Every calculation (overall mark, bands, outcome evidence, trends, calibration, CSV) is a pure TypeScript function in `apps/teaching/src/markbook/`. The DOM modules only draw what those functions return.

**Tech Stack:** TypeScript + Vite + vitest (jsdom) in `apps/teaching`; zod 3; Netlify Functions (`.mjs`) with Netlify Blobs; `node:test` for server tests in `tests/unit`; esbuild generator `scripts/generate-alchemy-schema.mjs`.

**Spec:** `docs/superpowers/specs/2026-10-22-markbook-design.md`

**Deviations from the spec, decided while planning:**
- **Endpoint path is `/api/classes/:id/markbook`.** The class id comes from the path, so request bodies don't carry `class_id`.
- **Clearing attempts doesn't exist on `main` yet.** Another session is adding it. The markbook reads `quiz/{lesson}/cleared/{owner}` → `{ from }` if it is there and ignores it if not, so it works before and after that change lands.
- **Question outcomes live in the published key.** A question's `outcomes` only reach the markbook after the quiz is republished. Until then the question uses its lesson's outcomes. The lesson's outcomes are `outcome_ids`, falling back to `syllabus_outcomes`, the same rule as `attachedOutcomeIds`.
- **A quiz cell opens a small attempt summary, not the Results tab.** The summary shows each question's marks and has a "Mark in Results" link to the lesson page. The Results tab has no deep link to one attempt, and building one is out of scope.
- **Outcomes are entered as codes.** A criterion's outcomes field and the quiz question's outcome field both take codes, comma-separated (`EN12-1, EN12-3`). Codes are matched to catalogue ids from the curriculum snapshot, and unknown codes are named inline. This avoids a large picker UI.
- **Bands are entered as text** per criterion, `A 13, B 10, C 7, D 4, E 0`.
- **Surname sort** uses the last word of the name. Ties fall back to the full name.
- **Removed students** show as "Removed student" rows when "Show removed students" is on, because their names are gone from the list.
- **Phone layout** switches at `max-width: 600px`.

```ts
import { z } from 'zod';

/**
 * A class markbook: the teacher's own columns and the marks typed into them.
 * Quiz columns are never stored here; the server builds them from attempts
 * each time the markbook opens.
 *
 * Generated to netlify/functions/_shared/markbook-schema.mjs for the server.
 */

export const MARKBOOK_LIMITS = { columns: 200, criteria: 12, bands: 6, outcomes: 8, title: 120, label: 120, band: 24, feedback: 2000, note: 200 } as const;

export const MarkbookIdSchema = z.string().regex(/^[a-z0-9]{8,24}$/);
const Day = z.string().regex(/^\d{4}-\d{2}-\d{2}$/, 'Pick a date.');
const Title = z.string().trim().min(1, 'Give it a title.').max(MARKBOOK_LIMITS.title);

export const BandSchema = z.object({ label: z.string().trim().min(1).max(MARKBOOK_LIMITS.band), min: z.number().min(0) });

export const CriterionSchema = z.object({
  id: MarkbookIdSchema,
  label: z.string().trim().min(1, 'Name each criterion.').max(MARKBOOK_LIMITS.label),
  max: z.number().positive('Each criterion needs marks.').max(1000),
  outcomes: z.array(z.string().min(1).max(64)).max(MARKBOOK_LIMITS.outcomes).default([]),
  bands: z.array(BandSchema).max(MARKBOOK_LIMITS.bands).optional()
});

export const AssessmentColumnSchema = z.object({
  id: MarkbookIdSchema,
  kind: z.literal('assessment'),
  title: Title,
  date: Day,
  weight: z.number().min(0).max(100),
  criteria: z.array(CriterionSchema).min(1, 'Add at least one criterion.').max(MARKBOOK_LIMITS.criteria)
});
export const HandinColumnSchema = z.object({ id: MarkbookIdSchema, kind: z.literal('handin'), title: Title, due: Day });
export const CheckColumnSchema = z.object({ id: MarkbookIdSchema, kind: z.literal('check'), title: Title, date: Day, scale: z.enum(['tick', 'four']) });
export const ColumnSchema = z.discriminatedUnion('kind', [AssessmentColumnSchema, HandinColumnSchema, CheckColumnSchema]);

export const HANDIN_STATUSES = ['submitted', 'late', 'missing', 'exempt'] as const;
export const AssessmentMarkSchema = z.object({
  criteria: z.record(z.string(), z.number().min(0)).default({}),
  feedback: z.string().trim().max(MARKBOOK_LIMITS.feedback).optional(),
  exempt: z.boolean().optional(),
  note: z.string().trim().max(MARKBOOK_LIMITS.note).optional()
});
export const HandinMarkSchema = z.object({ status: z.enum(HANDIN_STATUSES), at: Day.optional() });
export const CheckMarkSchema = z.union([z.boolean(), z.number().int().min(1).max(4)]);

export const MarkbookSchema = z.object({
  columns: z.array(ColumnSchema).max(MARKBOOK_LIMITS.columns).default([]),
  hidden_quizzes: z.array(z.string().min(1).max(64)).default([]),
  marks: z.record(z.string(), z.record(z.string(), z.unknown())).default({}),
  updated_at: z.string().default('')
});

export type Band = z.infer<typeof BandSchema>;
export type Criterion = z.infer<typeof CriterionSchema>;
export type AssessmentColumn = z.infer<typeof AssessmentColumnSchema>;
export type HandinColumn = z.infer<typeof HandinColumnSchema>;
export type CheckColumn = z.infer<typeof CheckColumnSchema>;
export type Column = z.infer<typeof ColumnSchema>;
export type AssessmentMark = z.infer<typeof AssessmentMarkSchema>;
export type HandinMark = z.infer<typeof HandinMarkSchema>;
export type HandinStatus = (typeof HANDIN_STATUSES)[number];
export type CheckMark = z.infer<typeof CheckMarkSchema>;
export type Mark = AssessmentMark | HandinMark | CheckMark;
export type Markbook = z.infer<typeof MarkbookSchema>;

const firstIssue = (error: z.ZodError): string => error.issues[0]?.message ?? 'Check this column.';

/** A column from the form or the wire, or the first thing to fix. */
export function checkColumn(value: unknown): { column: Column } | { problem: string } {
  const parsed = ColumnSchema.safeParse(value);
  if (!parsed.success) return { problem: firstIssue(parsed.error) };
  const column = parsed.data;
  if (column.kind === 'assessment') {
    if (new Set(column.criteria.map((c) => c.id)).size !== column.criteria.length) return { problem: 'Two criteria share an id.' };
    for (const c of column.criteria) {
      const high = (c.bands ?? []).find((b) => b.min > c.max);
      if (high) return { problem: `Band ${high.label} starts above ${c.max} in ${c.label}.` };
    }
  }
  return { column };
}

/** One student's mark for a column, or the first thing to fix. */
export function checkMark(column: Column, value: unknown): { mark: Mark } | { problem: string } {
  if (column.kind === 'assessment') {
    const parsed = AssessmentMarkSchema.safeParse(value);
    if (!parsed.success) return { problem: firstIssue(parsed.error) };
    const mark = parsed.data;
    if (mark.exempt) return { mark: { criteria: {}, exempt: true, ...(mark.note ? { note: mark.note } : {}) } };
    for (const [id, given] of Object.entries(mark.criteria)) {
      const criterion = column.criteria.find((c) => c.id === id);
      if (!criterion) return { problem: 'Unknown criterion.' };
      if (given > criterion.max) return { problem: `Give between 0 and ${criterion.max} for ${criterion.label}.` };
    }
    return { mark };
  }
  if (column.kind === 'handin') {
    const parsed = HandinMarkSchema.safeParse(value);
    return parsed.success ? { mark: parsed.data } : { problem: 'Pick submitted, late, missing or exempt.' };
  }
  if (column.scale === 'tick') return typeof value === 'boolean' ? { mark: value } : { problem: 'Tick or cross only.' };
  return typeof value === 'number' && Number.isInteger(value) && value >= 1 && value <= 4 ? { mark: value } : { problem: 'Give 1 to 4.' };
}

/** After a column is edited, keep only the marks that still fit it. */
export function pruneMarks(column: Column, marks: Record<string, unknown> = {}): Record<string, unknown> {
  const out: Record<string, unknown> = {};
  for (const [studentId, value] of Object.entries(marks)) {
    if (column.kind === 'assessment' && value && typeof value === 'object') {
      const mark = value as AssessmentMark;
      const ids = new Set(column.criteria.map((c) => c.id));
      out[studentId] = { ...mark, criteria: Object.fromEntries(Object.entries(mark.criteria ?? {}).filter(([id]) => ids.has(id))) };
      continue;
    }
    const checked = checkMark(column, value);
    if ('mark' in checked) out[studentId] = checked.mark;
  }
  return out;
}

const ALPHABET = 'abcdefghijklmnopqrstuvwxyz0123456789';
export function newMarkbookId(random: () => number = Math.random): string {
  let id = '';
  for (let i = 0; i < 12; i += 1) id += ALPHABET[Math.floor(random() * ALPHABET.length)];
  return id;
}
```

```ts
/** Plain sums behind the markbook. No DOM, so tests and every view share them. */

export type AssessmentResult = { exempt: boolean; score: number | null; max: number; percent: number | null; complete: boolean };

export const criteriaMax = (column: AssessmentColumn): number => column.criteria.reduce((sum, c) => sum + c.max, 0);

export function assessmentResult(column: AssessmentColumn, mark: AssessmentMark | undefined): AssessmentResult {
  const max = criteriaMax(column);
  if (mark?.exempt) return { exempt: true, score: null, max, percent: null, complete: false };
  const given = column.criteria.map((c) => mark?.criteria?.[c.id]).filter((v): v is number => typeof v === 'number');
  if (!given.length) return { exempt: false, score: null, max, percent: null, complete: false };
  const score = Math.round(given.reduce((sum, v) => sum + v, 0) * 100) / 100;
  const complete = given.length === column.criteria.length;
  return { exempt: false, score, max, percent: complete ? (score / max) * 100 : null, complete };
}

/** The marking-guideline band a criterion mark falls in, or null without bands. */
export function bandFor(criterion: Criterion, value: number): string | null {
  const bands = [...(criterion.bands ?? [])].sort((a, b) => b.min - a.min);
  return bands.find((b) => value >= b.min)?.label ?? null;
}

  return book.marks[columnId]?.[studentId] as T | undefined;
}

export type Overall = { percent: number | null; partial: boolean };

/**
 * Weighted assessment mark. An exempt task drops out, so its weight spreads
 * over the rest; an unmarked task makes the result "so far".
 */
export function overallFor(book: Markbook, studentId: string): Overall {
  let weighted = 0;
  let weights = 0;
  let partial = false;
  for (const column of assessments(book)) {
    if (column.weight === 0) continue;
    const r = assessmentResult(column, markOf<AssessmentMark>(book, column.id, studentId));
    if (r.exempt) continue;
    if (r.percent === null) {
      partial = true;
      continue;
    }
    weighted += column.weight * r.percent;
    weights += column.weight;
  }
  return { percent: weights ? weighted / weights : null, partial };
}
```

```ts
import type { AssessmentMark } from './schema';
import type { MarkbookView, QuizColumn, QuizRow } from './types';
import { assessments, dateOf, markOf, quizPercent } from './calc';

/** Outcome evidence and quiz flags, worked out from what the markbook already holds. */

export type Evidence = { outcome: string; percent: number; source: string; item: string };

/** Every percentage that counts as evidence for an outcome, for one student. */
export function evidenceFor(view: MarkbookView, studentId: string): Evidence[] {
  const out: Evidence[] = [];
  for (const column of assessments(view.markbook)) {
    const mark = markOf<AssessmentMark>(view.markbook, column.id, studentId);
    if (!mark || mark.exempt) continue;
    for (const c of column.criteria) {
      const given = mark.criteria?.[c.id];
      if (typeof given !== 'number') continue;
      for (const outcome of c.outcomes) out.push({ outcome, percent: (given / c.max) * 100, source: column.title, item: c.label });
    }
  }
  for (const quiz of view.quizzes) {
    if (quiz.hidden) continue;
    const s = quiz.scores[studentId];
    if (!s) continue;
    for (const row of s.rows) {
      if (row.awarded === null || row.max <= 0) continue;
      const prompt = quiz.questions.find((q) => q.id === row.id)?.prompt || 'Question';
      for (const outcome of row.outcomes) out.push({ outcome, percent: (row.awarded / row.max) * 100, source: quiz.title, item: prompt });
    }
  }
  return out;
}

export function outcomeCell(evidence: Evidence[], outcome: string): { mean: number | null; count: number; items: Evidence[] } {
  const items = evidence.filter((e) => e.outcome === outcome);
  return { mean: items.length ? items.reduce((s, e) => s + e.percent, 0) / items.length : null, count: items.length, items };
}

/** Outcomes the markbook has any place for, criteria first. */
export function outcomesInView(view: MarkbookView): string[] {
  const seen = new Set<string>();
  for (const column of assessments(view.markbook)) for (const c of column.criteria) for (const o of c.outcomes) seen.add(o);
  for (const quiz of view.quizzes) {
    if (quiz.hidden) continue;
    for (const q of quiz.questions) for (const o of q.outcomes) seen.add(o);
    for (const s of Object.values(quiz.scores)) for (const r of s.rows) for (const o of r.outcomes) seen.add(o);
  }
  return [...seen];
}

export type Band4 = 'low' | 'mid' | 'good' | 'high';
export const band4 = (percent: number): Band4 => (percent < 50 ? 'low' : percent < 70 ? 'mid' : percent < 85 ? 'good' : 'high');

/** Share of marks awarded for one question, or null while it waits for the teacher. */
export const heatShare = (row: QuizRow | undefined): number | null => (!row || row.awarded === null || row.max <= 0 ? null : row.awarded / row.max);

/** Questions the class averaged below half on, lowest first. */
export function reteachList(quiz: QuizColumn, studentIds: string[]): Array<{ id: string; prompt: string; average: number }> {
  const out: Array<{ id: string; prompt: string; average: number }> = [];
  for (const q of quiz.questions) {
    let got = 0;
    let max = 0;
    for (const id of studentIds) {
      const row = quiz.scores[id]?.rows.find((r) => r.id === q.id);
      if (!row || row.awarded === null) continue;
      got += row.awarded;
      max += row.max;
    }
    if (max > 0 && got / max < 0.5) out.push({ id: q.id, prompt: q.prompt || 'Question', average: got / max });
  }
  return out.sort((a, b) => a.average - b.average);
}

/** A student's quiz percentages, oldest first, leaving out hidden quizzes and ones still being marked. */
export function quizSeries(view: MarkbookView, studentId: string): number[] {
  return view.quizzes
    .filter((q) => !q.hidden)
    .sort((a, b) => dateOf(a).localeCompare(dateOf(b)))
    .map((q) => q.scores[studentId])
    .filter((s) => s && s.pending === 0)
    .map((s) => quizPercent(s))
    .filter((p): p is number => p !== null);
}

const mean = (xs: number[]): number => xs.reduce((s, x) => s + x, 0) / xs.length;

/** Five or more quizzes, and the last three average 15 or more points below the ones before. */
export function isDropping(series: number[]): boolean {
  if (series.length < 5) return false;
  return mean(series.slice(0, -3)) - mean(series.slice(-3)) >= 15;
}

/** Three or more most-confident answers scored zero, and they are 40% or more of the student's most-confident answers. */
export function isConfidentlyWrong(view: MarkbookView, studentId: string): boolean {
  let sure = 0;
  let wrong = 0;
  for (const quiz of view.quizzes) {
    if (quiz.hidden || !quiz.confidence) continue;
    for (const row of quiz.scores[studentId]?.rows ?? []) {
      if (row.confidence !== 3 || row.awarded === null) continue;
      sure += 1;
      if (row.awarded === 0) wrong += 1;
    }
  }
  return wrong >= 3 && wrong / sure >= 0.4;
}
```

```ts
import type { Band } from './schema';
import type { OutcomeLabel } from './types';

/** Short text fields the teacher types instead of using pickers: bands and outcome codes. */

export function parseBands(text: string): { bands: Band[] } | { problem: string } {
  const parts = text.split(',').map((p) => p.trim()).filter(Boolean);
  const bands: Band[] = [];
  for (const part of parts) {
    const match = /^(.+?)\s+(\d+(?:\.\d+)?)$/.exec(part);
    if (!match) return { problem: 'Write each band as a name then its lowest mark, like A 13.' };
    bands.push({ label: match[1]!.trim(), min: Number(match[2]) });
  }
  return { bands };
}

export const formatBands = (bands: Band[] | undefined): string => (bands ?? []).map((b) => `${b.label} ${b.min}`).join(', ');

/** Codes or ids, separated by commas or spaces. Unknown codes come back so the form can name them. */
export function parseOutcomeCodes(text: string, labels: OutcomeLabel[]): { ids: string[]; unknown: string[] } {
  const ids: string[] = [];
  const unknown: string[] = [];
  for (const token of text.split(/[\s,]+/).map((t) => t.trim()).filter(Boolean)) {
    const found = labels.find((l) => l.code.toLowerCase() === token.toLowerCase() || l.id === token);
    if (found) {
      if (!ids.includes(found.id)) ids.push(found.id);
    } else unknown.push(token.toUpperCase());
  }
  return { ids, unknown };
}

export const formatOutcomeCodes = (ids: string[], labels: OutcomeLabel[]): string =>
  ids.map((id) => labels.find((l) => l.id === id)?.code ?? id).join(', ');

export function outcomeLabels(rows: ReadonlyArray<{ id: string; code: string; title: string; group: string; status?: string }>): OutcomeLabel[] {
  return rows.filter((r) => r.status !== 'trashed').map(({ id, code, title, group }) => ({ id, code, title, group }));
}
```

```ts
import type { AssessmentMark, CheckMark, HandinMark } from './schema';
import type { AnyColumn, MarkbookView } from './types';
import { assessmentResult, formatPercent, markOf, orderedColumns, overallFor, quizPercent, sortStudents } from './calc';
import { HANDIN_LABEL } from './cells';

const cell = (text: string): string => (/[",\n]/.test(text) ? `"${text.replace(/"/g, '""')}"` : text);
const blankDash = (text: string): string => (text === '–' ? '' : text);

function value(view: MarkbookView, column: AnyColumn, studentId: string): string {
  if (column.kind === 'quiz') return blankDash(formatPercent(quizPercent(column.scores[studentId])));
  if (column.kind === 'assessment') {
    const r = assessmentResult(column, markOf<AssessmentMark>(view.markbook, column.id, studentId));
    return r.exempt ? 'Exempt' : blankDash(formatPercent(r.percent));
  }
  if (column.kind === 'handin') {
    const mark = markOf<HandinMark>(view.markbook, column.id, studentId);
    return mark ? HANDIN_LABEL[mark.status] : '';
  }
  const mark = markOf<CheckMark>(view.markbook, column.id, studentId);
  return mark === true ? 'Yes' : mark === false ? 'No' : typeof mark === 'number' ? String(mark) : '';
}

/** One row per student on the list, every shown column, then the overall mark. */
export function markbookCsv(view: MarkbookView): string {
  const columns = orderedColumns(view);
  const lines = [['Initials', 'Name', ...columns.map((c) => c.title), 'Overall']];
  for (const s of sortStudents(view.students)) {
    lines.push([s.initials, s.name, ...columns.map((c) => value(view, c, s.id)), blankDash(formatPercent(overallFor(view.markbook, s.id).percent))]);
  }
  return lines.map((row) => row.map(cell).join(',')).join('\n');
}

export function csvFileName(className: string, today: string): string {
  const slug = className.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '') || 'class';
  return `markbook-${slug}-${today}.csv`;
}
```

```js
/** A lesson's own outcomes: canonical ids first, else its older outcome codes. */
const lessonOutcomes = (draft) => (draft.outcome_ids?.length ? draft.outcome_ids : draft.syllabus_outcomes ?? []);

/**
 * Each listed student's first handed-in attempt at this quiz in one class,
 * scored as the Results tab scores it. Null when the lesson isn't a published
 * quiz. Goes before a teacher's clear (`quiz/{lesson}/cleared/{owner}`) are
 * skipped.
 */
export async function classQuizScores(store, lessonId, classId, studentIds) {
  let loaded;
  try {
    loaded = await loadQuiz(store, lessonId);
  } catch (error) {
    if (error instanceof QuizError) return null;
    throw error;
  }
  const { draft, record, settings, k } = loaded;
  const fallback = lessonOutcomes(draft);
  const outcomesOf = (rec, id) => entryFor(rec, id)?.key.outcomes ?? fallback;
  const owners = new Map(studentIds.map((id) => [deviceHash(`student:${classId}:${id}`), id]));
  const handedIn = new Map();
  for (const key of await listBlobKeys(store, `quiz/${lessonId}/results/`)) {
    const [, , , owner, nText] = key.split('/');
    if (owners.has(owner)) handedIn.set(owner, [...(handedIn.get(owner) ?? []), Number(nText)]);
  }
  const scores = {};
  let firstAt = null;
  await Promise.all(
    [...handedIn].map(async ([owner, ns]) => {
      const from = (await read(store, `quiz/${lessonId}/cleared/${owner}`))?.from ?? 0;
      for (const n of ns.filter((x) => x >= from).sort((a, b) => a - b)) {
        const [start, result] = await Promise.all([read(store, k.attempt(owner, n)), read(store, k.result(owner, n))]);
        if (!start || !result) continue;
        const rec = await recordFor(store, lessonId, start.version, record);
        const scored = scoreAttempt(rec, result, await read(store, k.marks(owner, n)));
        scores[owners.get(owner)] = {
          attempt_id: attemptId(owner, n),
          submitted_at: result.submitted_at,
          score: scored.score,
          max: scored.max,
          pending: scored.pending,
          rows: scored.rows.map((r) => ({ id: r.id, awarded: r.awarded, max: r.mark.max, confidence: r.confidence, outcomes: outcomesOf(rec, r.id) }))
        };
        if (!firstAt || result.submitted_at < firstAt) firstAt = result.submitted_at;
        return;
      }
    })
  );
  return {
    lesson_id: lessonId,
    title: draft.title,
    confidence: !!settings.confidence,
    first_at: firstAt,
    questions: record.order.map((id) => ({ id, prompt: record.questions[id].question.prompt, marks: record.questions[id].key.marks, outcomes: outcomesOf(record, id) })),
    scores
  };
}
```

```js
 * A class markbook, for the teacher only.
 *
 *   markbook/{class}   { columns, hidden_quizzes, marks: { column: { student: mark } }, updated_at }
 *
 * Quiz columns are not stored. They are built on every load from the class's
 * quiz attempts, so a question marked later in Results shows at once.
 */

import { isDeletedRecord } from './record-liveness.mjs';
import { unitKey } from './teaching-blobs.mjs';
import { loadRoster } from './class-claims.mjs';
import { classQuizScores } from './quiz.mjs';

const bookKey = (classId) => `markbook/${classId}`;
const sydneyDay = (iso) => new Intl.DateTimeFormat('en-CA', { timeZone: 'Australia/Sydney', year: 'numeric', month: '2-digit', day: '2-digit' }).format(new Date(iso));

async function readBook(store, classId) {
  const row = await store.getWithMetadata(bookKey(classId), { type: 'json', consistency: 'strong' });
  return { book: MarkbookSchema.parse(row?.data ?? {}), etag: row?.etag ?? null };
}

/** Read, change, write only if nobody wrote in between. One retry, then a 409. */
async function changeBook(store, classId, fn) {
  for (let i = 0; i < 2; i += 1) {
    const { book, etag } = await readBook(store, classId);
    const next = fn(book);
    next.updated_at = new Date().toISOString();
    const written = await store.setJSON(bookKey(classId), next, etag ? { onlyIfMatch: etag } : { onlyIfNew: true });
    if (written?.modified !== false) return next;
  }
  reject(409, 'changed', 'The markbook changed in another tab. Reload to see it.');
}

/** Quiz lessons in the class's active units that someone on the list has handed in. */
export async function quizColumns(store, roster, book) {

    if (!unit || isDeletedRecord(unit)) continue;

    }));
}

export async function markbookAction(store, classId, body) {
  if (!body || typeof body !== 'object' || Array.isArray(body)) reject(400, 'bad_request', 'Invalid request.');
  const roster = await loadRoster(store, classId);
  if (!roster) reject(404, 'not_found', 'Class not found.');
  const action = body.action ?? 'load';

  if (action === 'load') {
    const { book } = await readBook(store, classId);
    return {
      markbook: book,
      students: roster.students.map(({ id, name, initials }) => ({ id, name, initials })),
      quizzes: await quizColumns(store, roster, book)
    };
  }

  if (action === 'save_column') {
    const checked = checkColumn(body.column);
    if ('problem' in checked) reject(400, 'bad_column', checked.problem);
    const column = checked.column;
    const markbook = await changeBook(store, classId, (book) => {
      const at = book.columns.findIndex((c) => c.id === column.id);
      if (at === -1) {
        if (book.columns.length >= MARKBOOK_LIMITS.columns) reject(400, 'too_many', 'This markbook is full.');
        book.columns.push(column);
      } else {
        if (book.columns[at].kind !== column.kind) reject(400, 'bad_column', 'A column can’t change kind.');
        book.columns[at] = column;
        book.marks[column.id] = pruneMarks(column, book.marks[column.id]);
      }
      return book;
    });
    return { markbook };
  }

  if (action === 'delete_column') {
    const markbook = await changeBook(store, classId, (book) => {
      if (!book.columns.some((c) => c.id === body.column_id)) reject(404, 'not_found', 'That column is gone.');
      book.columns = book.columns.filter((c) => c.id !== body.column_id);
      delete book.marks[body.column_id];
      return book;
    });
    return { markbook };
  }

  if (action === 'set_mark') {
    if (!roster.students.some((s) => s.id === body.student_id)) reject(404, 'not_found', 'That student isn’t on the class list.');
    const markbook = await changeBook(store, classId, (book) => {
      const column = book.columns.find((c) => c.id === body.column_id);
      if (!column) reject(404, 'not_found', 'That column is gone.');
      const marks = { ...(book.marks[column.id] ?? {}) };
      if (body.mark === null) delete marks[body.student_id];
      else {
        const checked = checkMark(column, body.mark);
        if ('problem' in checked) reject(400, 'bad_mark', checked.problem);
        marks[body.student_id] = checked.mark;
      }
      book.marks[column.id] = marks;
      return book;
    });
    return { markbook };
  }

  if (action === 'hide_quiz') {
    if (!isSafeId(body.lesson_id)) reject(400, 'bad_request', 'Unknown quiz.');
    const markbook = await changeBook(store, classId, (book) => {
      const rest = book.hidden_quizzes.filter((id) => id !== body.lesson_id);
      book.hidden_quizzes = body.hidden === true ? [...rest, body.lesson_id] : rest;
      return book;
    });
    return { markbook };
  }

  reject(400, 'bad_action', 'Unknown action.');
}
```

```js
import { markbookAction } from './_shared/markbook.mjs';

/** Teacher only: a class's markbook. */
export function createMarkbookHandler(deps = {}) {
  return createOperatorHandler(async (request, context) => {
    const { env, store } = context;
    if (request.method !== 'POST') return withCors(methodNotAllowed('POST, OPTIONS'), request, env);
    const originError = guardRequestOrigin(request, env);
    if (originError) return withCors(originError, request, env);
    const match = new URL(request.url).pathname.match(/\/classes\/([^/]+)\/markbook$/);
    try {
      const result = await markbookAction(store, match?.[1], await request.json());
      return withCors(okResponse(200, result, { 'cache-control': 'no-store' }), request, env);
    } catch (e) {
      if (e instanceof QuizError || e instanceof SyntaxError) return withCors(errorResponse(e.status ?? 400, e.code ?? 'bad_request', e instanceof SyntaxError ? 'Invalid request.' : e.message, false), request, env);
      throw e;
    }
  }, deps);
}
```

```ts
  const setLocalMark = (columnId: string, studentId: string, mark: unknown | null): void => {
    if (!view) return;
    const byStudent = { ...(view.markbook.marks[columnId] ?? {}) };
    if (mark === null) delete byStudent[studentId];
    else byStudent[studentId] = mark;
    view = { ...view, markbook: { ...view.markbook, marks: { ...view.markbook.marks, [columnId]: byStudent } } };
  };

  /** Writes go one at a time; the server's copy is shown once the last one lands. */
  function save(run: () => Promise<Markbook>): void {
    pending += 1;
    queue = queue.then(async () => {
      try {
        const markbook = await run();
        pending -= 1;
        if (view && pending === 0) view = { ...view, markbook };
        status.textContent = '';
      } catch (error) {
        pending -= 1;
        status.textContent = error instanceof Error ? error.message : 'Couldn’t save that.';
        await load();
      }
      paint();
    });
  }

  const actions: MarkbookActions = {
    setMark: (columnId, studentId, mark) => {
      setLocalMark(columnId, studentId, mark);
      paint();
      save(() => api.setMark(cls.id, columnId, studentId, mark));
    },
    hideQuiz: (lessonId, hidden) => {
      if (view) view = { ...view, quizzes: view.quizzes.map((q) => (q.lesson_id === lessonId ? { ...q, hidden } : q)) };
      paint();
      save(() => api.hideQuiz(cls.id, lessonId, hidden));
    },
    editColumn: () => { status.textContent = 'Column editing arrives in the next step.'; },
    openAssessment: () => { status.textContent = 'Assessment marking arrives in the next step.'; },
    openQuiz: () => { status.textContent = 'Attempt summaries arrive in the next step.'; },
    openHeatmap: () => { status.textContent = 'Heatmaps arrive in the next step.'; }
  };

  function paint(): void {
    if (!view) return;
    const total = weightTotal(view.markbook);
    warning.hidden = total === 100 || assessments(view.markbook).length === 0;
    warning.textContent = `Assessment weights add to ${total}%, not 100%.`;
    toggle.textContent = mode === 'grid' ? 'Outcomes' : 'Marks';
    toggle.setAttribute('aria-pressed', String(mode === 'outcomes'));
    removedToggle.hidden = removedStudentIds(view).length === 0;
    removedToggle.textContent = showRemoved ? 'Hide removed students' : 'Show removed students';
    if (!view.students.length) {
      body.replaceChildren(el('p', 'mb__empty', 'Add students to this class to start a markbook.'));
      return;
    }
    const students = rowStudents(view, showRemoved);
    body.replaceChildren(renderGrid(view, students, actions, today()));
  }

  async function load(): Promise<void> {
    try {
      view = await api.load(cls.id);
      paint();
    } catch {
      status.textContent = 'Couldn’t load the markbook.';
    }
  }

  toggle.addEventListener('click', () => {
    mode = mode === 'grid' ? 'outcomes' : 'grid';
    paint();
  });
  removedToggle.addEventListener('click', () => {
    showRemoved = !showRemoved;
    paint();
  });
  const onMedia = () => paint();
  media?.addEventListener('change', onMedia);
  void load();
  void phoneIndex;
  void narrow;
  return () => media?.removeEventListener('change', onMedia);
}
```

```ts
  l.append(el('span', 'mb-form__label', text), control);
  return l;

export function openColumnForm(o: ColumnFormOptions): HTMLElement {
  const random = o.random ?? Math.random;
  const existing = o.column;
  const overlay = el('div', 'mb-overlay');
  const form = el('form', 'mb-form');
  form.setAttribute('role', 'dialog');
  form.setAttribute('aria-modal', 'true');
  form.setAttribute('aria-label', `${existing ? 'Edit' : 'Add'} ${KIND_TITLE[o.kind].toLowerCase()}`);
  form.noValidate = true;

  const date = input('date', existing ? (existing.kind === 'handin' ? existing.due : existing.date) : o.today, 'date');
  form.append(el('h3', 'mb-form__title', `${existing ? 'Edit' : 'Add'} ${KIND_TITLE[o.kind].toLowerCase()}`), labelled('Title', title), labelled(o.kind === 'handin' ? 'Due' : 'Date', date));

    opt.value = value;
    scale.append(opt);

  const blankCriterion = (): Criterion => ({ id: newMarkbookId(random), label: '', max: 10, outcomes: [] });
  let criteria: Criterion[] = existing?.kind === 'assessment' ? existing.criteria.map((c) => ({ ...c })) : [blankCriterion()];
  const drafts = new Map<string, { label: HTMLInputElement; max: HTMLInputElement; outcomes: HTMLInputElement; bands: HTMLInputElement }>();

  function paintCriteria(): void {
    criteriaHost.replaceChildren();
    drafts.clear();
    for (const c of criteria) {
      const row = el('fieldset', 'mb-form__criterion');
      row.append(el('legend', 'mb-form__label', 'Criterion'));
      const fields = {
        label: input('criterion-label', c.label),
        max: input('criterion-max', String(c.max), 'number'),
        outcomes: input('criterion-outcomes', formatOutcomeCodes(c.outcomes, o.outcomes)),

      remove.type = 'button';
      remove.disabled = criteria.length === 1;
      remove.addEventListener('click', () => {
        criteria = criteria.filter((x) => x.id !== c.id);
        paintCriteria();
      });
      row.append(labelled('Name', fields.label), labelled('Marks', fields.max), labelled('Outcome codes', fields.outcomes), labelled('Bands (optional)', fields.bands), remove);
      criteriaHost.append(row);

    add.type = 'button';
    add.addEventListener('click', () => {
      criteria = [...readCriteriaLoose(), blankCriterion()];
      paintCriteria();
    });
    paintCriteria();
    form.append(labelled('Weight (% of the course mark)', weight), criteriaHost, add);
  }
  if (o.kind === 'check') form.append(labelled('Scale', scale));

  /** Keep typed text when the criteria list is redrawn. */
  function readCriteriaLoose(): Criterion[] {
    return criteria.map((c) => {
      const f = drafts.get(c.id);
      if (!f) return c;
      const bands = parseBands(f.bands.value);
      return { ...c, label: f.label.value, max: Number(f.max.value) || 0, outcomes: parseOutcomeCodes(f.outcomes.value, o.outcomes).ids, bands: 'bands' in bands ? bands.bands : c.bands };
    });
  }

  const error = el('p', 'mb-form__error');
  error.setAttribute('role', 'alert');

  function build(): Column | string {
    const id = existing?.id ?? newMarkbookId(random);
    if (o.kind === 'handin') return { id, kind: 'handin', title: title.value, due: date.value };
    if (o.kind === 'check') return { id, kind: 'check', title: title.value, date: date.value, scale: scale.value === 'four' ? 'four' : 'tick' };
    const built: Criterion[] = [];
    for (const c of criteria) {
      const f = drafts.get(c.id)!;
      const codes = parseOutcomeCodes(f.outcomes.value, o.outcomes);
      if (codes.unknown.length) return `${codes.unknown.join(', ')} isn’t an outcome code.`;
      const bands = parseBands(f.bands.value);
      if ('problem' in bands) return bands.problem;
      built.push({ id: c.id, label: f.label.value, max: Number(f.max.value), outcomes: codes.ids, ...(bands.bands.length ? { bands: bands.bands } : {}) });
    }
    return { id, kind: 'assessment', title: title.value, date: date.value, weight: Number(weight.value), criteria: built };
  }

  const actions = el('div', 'mb-form__actions');
  const save = el('button', 'btn btn--primary mb-form__save', 'Save');
  save.type = 'button';
  save.addEventListener('click', async () => {
    const draft = build();
    if (typeof draft === 'string') {
      error.textContent = draft;
      return;
    }
    const checked = checkColumn(draft);
    if ('problem' in checked) {
      error.textContent = checked.problem;
      return;
    }
    error.textContent = '';
    save.disabled = true;
    try {
      await o.onSave(checked.column);
      close();
    } catch (e) {
      error.textContent = e instanceof Error ? e.message : 'Couldn’t save that.';
      save.disabled = false;
    }
  });
  const cancel = el('button', 'btn btn--secondary', 'Cancel');
  cancel.type = 'button';
  cancel.addEventListener('click', close);
  actions.append(save, cancel);
  if (existing && o.onDelete) {
    const del = el('button', 'btn btn--secondary mb-form__delete', 'Delete column');
    del.type = 'button';
    del.addEventListener('click', () => {
      if (typeof window.confirm === 'function' && !window.confirm(`Delete ${existing.title} and every mark in it?`)) return;
      o.onDelete?.();
      close();
    });
    actions.append(del);
  }
  form.append(error, actions);
  form.addEventListener('keydown', (event) => {
    if (event.key === 'Escape') close();
  });
  overlay.append(form);
  document.body.append(overlay);
  title.focus();
  return overlay;
}
```

```ts
import { el } from '@/quiz/dom';
import type { AssessmentColumn, AssessmentMark } from './schema';
import type { RosterStudent } from './types';
import { bandFor } from './calc';

/**
 * Marking one assessment task, one student at a time, from a pile of scripts.
 * Enter saves and moves on, so the keyboard never has to leave the boxes.
 */

export type AssessmentPanelOptions = {
  column: AssessmentColumn;
  students: RosterStudent[];
  studentId: string;
  markFor: (studentId: string) => AssessmentMark | undefined;
  onSave: (studentId: string, mark: AssessmentMark | null) => void;
};

export function openAssessmentPanel(o: AssessmentPanelOptions): HTMLElement {
  const overlay = el('div', 'mb-overlay');
  const panel = el('div', 'mb-panel');
  panel.setAttribute('role', 'dialog');
  panel.setAttribute('aria-modal', 'true');
  panel.setAttribute('aria-label', `Mark ${o.column.title}`);
  overlay.append(panel);
  document.body.append(overlay);
  let current = o.studentId;
  let boxes: HTMLInputElement[] = [];
  let feedback: HTMLTextAreaElement;
  let exempt: HTMLInputElement;
  let note: HTMLInputElement;
  let error: HTMLElement;
  const close = () => overlay.remove();

  function paint(): void {
    const student = o.students.find((s) => s.id === current) ?? { id: current, name: 'Removed student', initials: '–' };
    const mark = o.markFor(current);
    panel.replaceChildren();
    panel.append(el('p', 'mb-panel__task', o.column.title), el('h3', 'mb-panel__student', `${student.initials} · ${student.name}`));
    boxes = o.column.criteria.map((c) => {
      const row = el('label', 'mb-panel__row');
      const box = el('input', 'mb-form__input mb-panel__mark');
      box.type = 'number';
      box.inputMode = 'decimal';
      box.min = '0';
      box.max = String(c.max);
      box.step = '0.5';
      const given = mark?.criteria?.[c.id];
      box.value = typeof given === 'number' ? String(given) : '';
      const band = el('span', 'mb-panel__band');
      const showBand = () => {
        const n = Number(box.value);
        const label = box.value.trim() && Number.isFinite(n) ? bandFor(c, n) : null;
        band.textContent = label ? `Band ${label}` : '';
      };
      box.addEventListener('input', showBand);
      showBand();
      box.addEventListener('keydown', (event) => {
        if (event.key === 'Enter') {
          event.preventDefault();
          saveAndNext();
        }
      });
      row.append(el('span', 'mb-form__label', `${c.label} (out of ${c.max})`), box, band);
      panel.append(row);
      return box;
    });
    feedback = el('textarea', 'mb-form__input mb-panel__feedback');
    feedback.rows = 3;
    feedback.value = mark?.feedback ?? '';
    const exemptRow = el('label', 'mb-panel__exempt-row');
    exempt = el('input', 'mb-panel__exempt');
    exempt.type = 'checkbox';
    exempt.checked = !!mark?.exempt;
    exemptRow.append(exempt, el('span', undefined, 'Exempt from this task'));
    note = el('input', 'mb-form__input mb-panel__note');
    note.placeholder = 'Reason or provision, e.g. misadventure, extra time';
    note.value = mark?.note ?? '';
    error = el('p', 'mb-form__error mb-panel__error');
    error.setAttribute('role', 'alert');
    const fb = el('label', 'mb-form__field');
    fb.append(el('span', 'mb-form__label', 'Feedback (optional)'), feedback);
    const nt = el('label', 'mb-form__field');
    nt.append(el('span', 'mb-form__label', 'Note'), note);
    const actions = el('div', 'mb-form__actions');
    const next = el('button', 'btn btn--primary mb-panel__next', 'Save and next');
    next.type = 'button';
    next.addEventListener('click', saveAndNext);
    const saveBtn = el('button', 'btn btn--secondary mb-panel__save', 'Save');
    saveBtn.type = 'button';
    saveBtn.addEventListener('click', () => {
      if (save()) close();
    });
    const closeBtn = el('button', 'btn btn--secondary', 'Close');
    closeBtn.type = 'button';
    closeBtn.addEventListener('click', close);
    actions.append(next, saveBtn, closeBtn);
    panel.append(fb, exemptRow, nt, error, actions);
  }

  function collect(): AssessmentMark | null | string {
    const noteText = note.value.trim();
    if (exempt.checked) return { criteria: {}, exempt: true, ...(noteText ? { note: noteText } : {}) };
    const criteria: Record<string, number> = {};
    for (const [i, c] of o.column.criteria.entries()) {
      const text = boxes[i]!.value.trim();
      if (!text) continue;
      const n = Number(text);
      if (!Number.isFinite(n) || n < 0 || n > c.max) return `Give between 0 and ${c.max} for ${c.label}.`;
      criteria[c.id] = n;
    }
    const fbText = feedback.value.trim();
    if (!Object.keys(criteria).length && !fbText && !noteText) return null;
    return { criteria, ...(fbText ? { feedback: fbText } : {}), ...(noteText ? { note: noteText } : {}) };
  }

  function save(): boolean {
    const mark = collect();
    if (typeof mark === 'string') {
      error.textContent = mark;
      return false;
    }
    o.onSave(current, mark);
    return true;
  }

  function saveAndNext(): void {
    if (!save()) return;
    const at = o.students.findIndex((s) => s.id === current);
    const next = o.students[at + 1];
    if (!next) {
      close();
      return;
    }
    current = next.id;
    paint();
    boxes[0]?.focus();
  }

  panel.addEventListener('keydown', (event) => {
    if (event.key === 'Escape') close();
  });
  paint();
  boxes[0]?.focus();
  return overlay;
}
```

```ts
function sheet(label: string): { overlay: HTMLElement; body: HTMLElement } {

  body.setAttribute('role', 'dialog');

  document.body.append(overlay);
  return { overlay, body };
}

const short = (text: string): string => (text.length > 60 ? `${text.slice(0, 57)}…` : text || 'Question');

export function openHeatmap(quiz: QuizColumn, students: RosterStudent[]): HTMLElement {
  const { overlay, body } = sheet(`${quiz.title} heatmap`);
  body.append(el('h3', 'mb-form__title', quiz.title));
  const reteach = reteachList(quiz, students.map((s) => s.id));
  body.append(el('h4', 'mb-form__label', 'Reteach'));
  if (!reteach.length) body.append(el('p', 'mb__empty', 'No question averaged below 50%.'));
  else {
    const list = el('ol', 'mb-reteach');
    for (const r of reteach) list.append(el('li', undefined, `${short(r.prompt)} · ${Math.round(r.average * 100)}%`));
    body.append(list);
  }
  const grid = el('div', 'mb-grid');
  const table = el('table', 'mb-grid__table mb-heat');
  const headRow = el('tr');
  const corner = el('th', 'mb-grid__student', 'Question');
  corner.scope = 'col';
  headRow.append(corner);
  for (const s of students) {
    const th = el('th', undefined, s.initials);
    th.scope = 'col';
    th.title = s.name;
    headRow.append(th);
  }
  const thead = el('thead');
  thead.append(headRow);
  const tbody = el('tbody');
  for (const q of quiz.questions) {
    const row = el('tr');
    const head = el('th', 'mb-grid__student', short(q.prompt));
    head.scope = 'row';
    row.append(head);
    for (const s of students) {
      const r = quiz.scores[s.id]?.rows.find((x) => x.id === q.id);
      const share = heatShare(r);
      const td = el('td', `mb-heat__cell ${!r ? 'mb-heat--none' : share === null ? 'mb-heat--pending' : `mb-band--${band4(share * 100)}`}`);
      td.setAttribute('aria-label', `${s.name}: ${!r ? 'not taken' : share === null ? 'to mark' : `${Math.round(share * 100)}%`}`);
      row.append(td);
    }
    tbody.append(row);
  }
  table.append(thead, tbody);
  grid.append(table);
  body.append(grid);
  return overlay;
}

export function openQuizAttempt(quiz: QuizColumn, student: RosterStudent, resultsHref: string): HTMLElement {
  const { overlay, body } = sheet(`${quiz.title}, ${student.name}`);
  body.append(el('p', 'mb-panel__task', quiz.title), el('h3', 'mb-panel__student', `${student.initials} · ${student.name}`));

  for (const q of quiz.questions) {
    const r = score?.rows.find((x) => x.id === q.id);
    list.append(el('li', undefined, `${short(q.prompt)} · ${!r ? 'not answered' : r.awarded === null ? 'to mark' : `${r.awarded}/${r.max}`}`));
  }
  const link = el('a', 'mb-attempt__link', 'Mark in Results');
  link.href = resultsHref;
  body.append(list, link);

}
```

```ts
/** At phone width, one student's card at a time, with the grid's own cell controls. */

export function renderPhone(view: MarkbookView, students: RosterStudent[], actions: MarkbookActions, index: number, move: (index: number) => void, today: string): HTMLElement {
  const at = Math.max(0, Math.min(students.length - 1, index));
  const s = students[at]!;
  const card = el('div', 'mb-phone');
  const nav = el('div', 'mb-phone__nav');
  const prev = el('button', 'btn btn--secondary mb-phone__prev', 'Previous');
  prev.type = 'button';
  prev.disabled = at === 0;
  prev.addEventListener('click', () => move(at - 1));
  const next = el('button', 'btn btn--secondary mb-phone__next', 'Next');
  next.type = 'button';
  next.disabled = at === students.length - 1;
  next.addEventListener('click', () => move(at + 1));
  const pick = el('select', 'mb-form__input mb-phone__pick');
  pick.setAttribute('aria-label', 'Student');
  students.forEach((x, i) => {
    const opt = el('option', undefined, `${x.initials} · ${x.name}`);
    opt.value = String(i);
    pick.append(opt);
  });
  pick.value = String(at);
  pick.addEventListener('change', () => move(Number(pick.value)));
  nav.append(prev, pick, next);
  const head = el('div', 'mb-phone__head');
  head.append(el('h3', 'mb-phone__name', `${s.initials} · ${s.name}`), el('span', 'mb-phone__overall', overallText(overallFor(view.markbook, s.id))));
  const list = el('div', 'mb-phone__list');
  for (const c of orderedColumns(view)) {
    const row = el('div', 'mb-phone__row');

    label.append(el('span', undefined, c.title), el('small', undefined, ` ${KIND_LABEL[c.kind]} · ${formatDay(dateOf(c))}`));
    row.append(label, cellFor(view, c, s, actions, today));
    list.append(row);
  }
  card.append(nav, head, list);
```
